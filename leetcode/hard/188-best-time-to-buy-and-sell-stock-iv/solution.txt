class Solution {
public:
    vector<vector<vector<int>>> cache;
    vector<int> arr;
    int f(int i, int held, int k) {
        if(i >= arr.size() || k <= 0) return 0;

        if(cache[i][k][held] != -1) return cache[i][k][held];
        int profit = 0;
        if(held == 0) {
            //buy it and held it
            profit = max(profit, -arr[i] + f(i + 1, 1, k));
            //wait for better opportunity
            profit = max(profit, f(i + 1, 0, k));
        } else { // hodling
            // keep holding
            profit = max(profit, f(i + 1, 1, k));
            // sell it and book profit
            profit = max(profit, arr[i] + f(i + 1, 0, k - 1));
        }
        return cache[i][k][held] = profit;
    }
