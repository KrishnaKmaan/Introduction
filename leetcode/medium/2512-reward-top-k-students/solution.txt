class Solution {
public:
    vector<int> topStudents(vector<string>& positive_feedback, vector<string>& negative_feedback, vector<string>& reports, vector<int>& student_id, int k) {
        unordered_set<string> pos(positive_feedback.begin(), positive_feedback.end());
        unordered_set<string> neg(negative_feedback.begin(), negative_feedback.end());

        map<int, int> mp;
        for(int i = 0; i < reports.size(); i++){
            string report = reports[i];
            stringstream ss(report);
            string word;
            mp[i] = 0;
            while(ss >> word) {
                if(pos.find(word) != pos.end()) {
                    mp[i] += 3;
                } else if(neg.find(word) != neg.end()) {
                    mp[i] -= 1;
                }
            }
        }

        vector<pair<int, int>> score;
