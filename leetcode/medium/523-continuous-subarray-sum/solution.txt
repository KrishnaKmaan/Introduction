class Solution {
public:
    bool checkSubarraySum(vector<int>& nums, int k) {
        unordered_map<int, int> reminder;
        int sum = 0;
        reminder[0] = -1;
        for(int i = 0; i < nums.size(); i++) {
            sum += nums[i];
            int rem = sum % k;
            if(reminder.find(rem) != reminder.end()) {
                if(i - reminder[rem] >= 2) return true;
            } else {
                reminder[rem] = i;
            }
        }
        return false;
    }
};
