class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        unordered_map<int, int> csumFreq;
        int n = nums.size();
        int cur = 0;
        int ans  = 0;
        csumFreq[0] = 1;
        for(int i = 0; i < n; i++) {
            cur += nums[i];
            ans += csumFreq[cur - k];
            csumFreq[cur]++;
        }
        return ans;
    }
};
